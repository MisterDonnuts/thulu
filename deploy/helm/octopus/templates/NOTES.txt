3. Agents run as pods in tenant-<workspace> namespaces. Label this namespace
   so the tenant NetworkPolicies (deploy/k8s-data-plane) let agent pods reach
   the daemon:
     kubectl label namespace {{ .Release.Namespace }} thulu.dev/plane=control
{{- if .Values.agentRuntime.admissionPolicy.enabled }}
   The admission policy "{{ include "octopus.fullname" . }}-daemon-scope" fences
   the daemon's ServiceAccount to those namespaces (Kubernetes >= 1.30).
{{- else }}

   WARNING: agentRuntime.admissionPolicy is disabled. The daemon's ClusterRole
   can create pods in ANY namespace, which is equivalent to reading every
   Secret in the cluster. Treat its credentials as cluster-admin.
{{- end }}
{{- end }}

{{- if and (eq .Values.agentRuntime.mode "docker") .Values.dockerSocket.enabled }}

3. SECURITY: this release hostPath-mounts the node's Docker socket so the daemon
   can launch agent containers. That grants root on the node. Pin the pod to a
   dedicated node (nodeSelector/taints) and keep it off shared nodes. For
   stronger isolation set dockerSocket.enabled=false and point DOCKER_HOST
   (via extraEnv) at an isolated Docker/sysbox endpoint.
{{- end }}

{{- if not .Values.secrets.existingSecret }}
{{- if not .Values.secrets.secretboxKey }}

   NOTE: secrets.secretboxKey is empty, so stored credentials (Git tokens,
   LLM keys) are encrypted with a key derived from the JWT secret. Set one
   (openssl rand -hex 32) and keep it; the daemon re-encrypts on the next start.
{{- end }}
{{- end }}

{{- if eq (lower (toString .Values.config.trustedProxies)) "private" }}

   WARNING: config.trustedProxies=private trusts X-Forwarded-For from every
   pod in the cluster, agent pods included, so any of them can forge the
   client address in the audit log and dodge per-IP rate limits. List the
   ingress controller's pod IPs or CIDR instead.
{{- else if not .Values.config.trustedProxies }}

   NOTE: config.trustedProxies is empty, so X-Forwarded-For is ignored and
   the audit log and rate limits see the ingress controller's address. Set it
   to the controller's pod CIDR (e.g. 10.42.0.0/24) to see real client IPs.
{{- end }}

{{- if and .Values.tls.enabled (not .Values.caBundle.configMap) }}

   WARNING: tls.enabled without caBundle.configMap. Agents and MCP servers
   dial the daemon over https and will reject its certificate unless a public
   CA issued it. Put the issuing CA in a ConfigMap and set caBundle.configMap.
{{- end }}
{{- if .Values.tls.enabled }}

     ingress.annotations."nginx\.ingress\.kubernetes\.io/backend-protocol"=HTTPS
{{- end }}
